cat << 'EOF' > setup_ai_agent.sh
#!/bin/bash
# -------------------------------------------------------------------
# Script d'installation automatisé Ollama + Coder LLM + Hermes Agent
# Optimisé pour Kali Linux (Environnement CPU / 16GB-32GB RAM)
# -------------------------------------------------------------------
set -e

echo "================================================================="
echo " 🚀 Initialisation de l'environnement IA pour Moad..."
echo "================================================================="

# 1. Mise à jour système et installation des dépendances clés
echo -e "\n--> [1/5] Installation des paquets requis (Python, Node, Build)..."
sudo apt update
sudo apt install -y curl git python3 python3-venv python3-pip build-essential nodejs npm

# 2. Installation d'Ollama
echo -e "\n--> [2/5] Installation d'Ollama..."
if ! command -v ollama &> /dev/null; then
    curl -fsSL https://ollama.com/install.sh | sh
else
    echo "Ollama est déjà installé."
fi

# S'assurer que le service Ollama tourne
sudo systemctl start ollama || true

# 3. Téléchargement et optimisation du meilleur LLM Codeur
echo -e "\n--> [3/5] Configuration du LLM (Qwen Coder)..."
echo "Téléchargement du modèle de base..."
ollama pull qwen2.5-coder:7b

echo "Création du Modelfile personnalisé pour contourner la limite de contexte d'Hermes..."
cat << 'MODEL_EOF' > Modelfile-hermes
FROM qwen2.5-coder:7b
# Ajustement du contexte à 64k requis au minimum par Hermes Agent
PARAMETER num_ctx 64000
MODEL_EOF

ollama create qwen-coder-hermes -f Modelfile-hermes

# 4. Installation de Hermes Agent
echo -e "\n--> [4/5] Clonage et build de Hermes Agent..."
mkdir -p ~/.hermes
cd ~/.hermes

if [ ! -d "hermes-agent" ]; then
    git clone https://github.com/NousResearch/Hermes-Agent.git hermes-agent
fi
cd hermes-agent

# Environnement virtuel Python pour Kali (évite le blocage 'externally-managed-environment')
python3 -m venv venv
source venv/bin/activate
pip install --upgrade pip setuptools wheel
pip install -e .

# Build de l'interface graphique (Frontend)
npm install
if npm run | grep -q "build"; then
    npm run build
fi

# 5. Génération de la configuration optimale (~/.hermes/config.yaml)
echo -e "\n--> [5/5] Création du fichier de configuration ~/.hermes/config.yaml..."
mkdir -p ~/.hermes

cat << 'CONFIG_EOF' > ~/.hermes/config.yaml
model:
  provider: ollama
  default: qwen-coder-hermes
  context_length: 64000
# Optimisation pour CPU Intel hybride (on cible la puissance brute)
ollama:
  options:
    num_thread: 4  # Ajusté pour ne pas saturer et bloquer sur les E-cores de l'i7-1265U
CONFIG_EOF

# Création du raccourci d'exécution global
mkdir -p ~/.local/bin
cat << 'LAUNCH_EOF' > ~/.local/bin/hermes
#!/bin/bash
source ~/.hermes/hermes-agent/venv/activate
exec python3 -m hermes_cli "$@"
LAUNCH_EOF
chmod +x ~/.local/bin/hermes

# Injection dans le PATH de Zsh
if [[ :$PATH: != *:"$HOME/.local/bin":* ]]; then
    echo 'export PATH="$HOME/.local/bin:$PATH"' >> ~/.zshrc
fi

echo "================================================================="
echo " 🎉 Tout est installé avec succès, Moad !"
echo "================================================================="
echo "Pour démarrer, exécute d'abord :"
echo "    source ~/.zshrc"
echo ""
echo "Ensuite, lance Hermes Agent avec :"
echo "    hermes chat"
echo "================================================================="
EOF

chmod +x setup_ai_agent.sh
